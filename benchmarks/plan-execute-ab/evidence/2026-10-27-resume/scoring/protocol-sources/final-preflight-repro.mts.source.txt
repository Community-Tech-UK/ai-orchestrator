import { existsSync, mkdirSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from 'fs';
import { tmpdir } from 'os';
import { join, resolve } from 'path';
import { prepareScoringDependencies, scoringEnvironment } from '../../benchmarks/plan-execute-ab/scoring-setup';
import { scoreRun } from '../../benchmarks/plan-execute-ab/scorer';
import { runCommand } from '../../benchmarks/plan-execute-ab/exec';
import type { TaskDefinition } from '../../benchmarks/plan-execute-ab/types';

const destination = resolve('benchmarks/plan-execute-ab/evidence/2026-10-27-resume/scoring/final-preflight');
mkdirSync(destination, { recursive: true });
const fixture = resolve('_scratch/lt-resume/final-native-blocked-fixture');
mkdirSync(fixture, { recursive: true });
const hostLock = JSON.parse(readFileSync('package-lock.json', 'utf8'));
const esbuild = hostLock.packages['node_modules/esbuild'];
const pkg = { name: 'native-boundary-fixture', version: '1.0.0', dependencies: { esbuild: esbuild.version }, scripts: { test: 'esbuild fixture.js --bundle --platform=node --outfile=bundle.cjs && node bundle.cjs' } };
writeFileSync(join(fixture, 'package.json'), JSON.stringify(pkg));
writeFileSync(join(fixture, 'package-lock.json'), JSON.stringify({ name: pkg.name, version: pkg.version, lockfileVersion: 3, packages: { '': pkg, ...Object.fromEntries(Object.entries(hostLock.packages).filter(([name]) => name === 'node_modules/esbuild' || name.startsWith('node_modules/@esbuild/'))) } }));
writeFileSync(join(fixture, 'fixture.js'), 'import { z } from "zod"; console.log(z.string().parse("HOST_DEPENDENCY_LOADED"));');
try {
  await prepareScoringDependencies(fixture);
  throw new Error('UNSAFE: native fixture setup was accepted');
} catch (error) {
  const receipt = { error: String(error), setupBlockedAtAncestorPreflight: String(error).includes('ancestor node_modules'), nodeModulesCreated: existsSync(join(fixture, 'node_modules')), dependencyReceiptCreated: existsSync(join(fixture, '_scratch/scoring-dependencies.json')) };
  writeFileSync(join(destination, 'native-bundler-blocked.json'), JSON.stringify(receipt, null, 2));
  if (!receipt.setupBlockedAtAncestorPreflight) throw error;
} finally { rmSync(fixture, { recursive: true, force: true }); }

const clean = mkdtempSync(join(tmpdir(), 'aio-scoring-pure-fixture-'));
const task: TaskDefinition = { id: 'FIXTURE', repoId: 'fixture', commitSha: 'fixed', parentSha: 'parent', commitMessage: '', size: 'small', prompt: 'Fixture', hiddenTests: ['fixture.cjs'], hiddenTestCommand: ['missing-benchmark-fixture-executable'], damageCommands: [], sourceFiles: [], verification: { failOnParent: true, passOnCommit: true, checkedAt: '', evidenceDir: '' } };
try {
  try { await scoreRun(task, { workspaceDir: clean }); throw new Error('UNSAFE: missing executable was scored'); }
  catch (error) {
    const receipt = { error: String(error), unscoredInfrastructureError: String(error).includes('scoring infrastructure failure') };
    writeFileSync(join(destination, 'missing-executable-unscored.json'), JSON.stringify(receipt, null, 2));
    if (!receipt.unscoredInfrastructureError) throw error;
  }
  const bootstrap = [];
  for (const command of [['node', '--version'], ['npm', '--version'], ['pnpm', '--version'], ['claude', '--version'], ['codex', '--version']]) {
    bootstrap.push(await runCommand(command, { cwd: clean, env: scoringEnvironment(clean), timeoutMs: 30_000 }));
  }
  writeFileSync(join(destination, 'global-cli-bootstrap.json'), JSON.stringify({ modelInvocations: 0, results: bootstrap }, null, 2));
  if (bootstrap.some(result => result.exitCode !== 0)) throw new Error('global CLI bootstrap failed');
} finally { rmSync(clean, { recursive: true, force: true }); }
console.log('Native fallback rejected before installation; missing executable unscored; five global CLI version probes passed.');
