// usage: node cdp.cjs <exprfile>  — evaluates an async expression in the first renderer page
const fs = require('node:fs');
const puppeteer = require('/Users/suas/work/orchestrat0r/ai-orchestrator/.worktrees/queue/2026-10-27-mimo-stale-session-42024f/node_modules/puppeteer-core');
const out = fs.readFileSync('/tmp/aio-lt-queue-5d42024f-tools/app.out', 'utf8');
const ws = out.match(/ws:\/\/127\.0\.0\.1:9507\/devtools\/browser\/[a-z0-9-]+/)[0];
(async () => {
  const browser = await puppeteer.connect({ browserWSEndpoint: ws, defaultViewport: null });
  const pages = (await browser.pages()).filter((p) => !p.url().startsWith('devtools'));
  const page = pages[0];
  const cdp = await page.createCDPSession();
  await cdp.send('Emulation.setFocusEmulationEnabled', { enabled: true }).catch(() => {});
  await cdp.send('Page.setWebLifecycleState', { state: 'active' }).catch(() => {});
  if (process.argv[2] === '--urls') { console.log(pages.map((p) => p.url())); await browser.disconnect(); return; }
  const expr = fs.readFileSync(process.argv[2], 'utf8');
  const r = await cdp.send('Runtime.evaluate', { expression: `(async()=>{${expr}})()`, awaitPromise: true, returnByValue: true });
  if (process.argv[3]) await page.screenshot({ path: process.argv[3] });
  console.log(JSON.stringify(r.result.value ?? r.exceptionDetails ?? r.result, null, 1));
  await browser.disconnect();
})().catch((e) => { console.error('ERR', e.message); process.exit(1); });
